## 🧼 Step 3: Warm-up Code to Pull a Wallet’s Transactions

Here’s a simple script that:
- Talks to the Etherscan API
- Pulls all normal transactions from a wallet
- Turns them into a pandas DataFrame

```python
import requests
import pandas as pd
from datetime import datetime

# Replace this with your own API key
API_KEY = "your_api_key_here"

# You can test with this random wallet or your own
wallet_address = "0xde0B295669a9FD93d5F28D9Ec85E40f4cb697BAe"

# Step 1: Fetch transactions from Etherscan
def fetch_wallet_transactions(address, api_key):
    url = "https://api.etherscan.io/api"
    params = {
        "module": "account",
        "action": "txlist",
        "address": address,
        "startblock": 0,
        "endblock": 99999999,
        "sort": "asc",
        "apikey": api_key
    }

    response = requests.get(url, params=params)
    data = response.json()

    # Turn the transaction list into a pandas DataFrame
    txs = data["result"]
    df = pd.DataFrame(txs)

    # Clean up the data
    df["timeStamp"] = pd.to_datetime(df["timeStamp"], unit="s")
    df["value"] = pd.to_numeric(df["value"]) / 1e18  # Convert Wei to ETH
    df["gasUsed"] = pd.to_numeric(df["gasUsed"])
    df["gasPrice"] = pd.to_numeric(df["gasPrice"]) / 1e9  # Gwei
    df["gasCost(ETH)"] = df["gasUsed"] * df["gasPrice"] / 1e9

    return df

# Step 2: Call the function and print a preview
df = fetch_wallet_transactions(wallet_address, API_KEY)
print(df[["hash", "timeStamp", "from", "to", "value", "gasCost(ETH)"]].head())
```
