Delete everything currently in the Shader Text box. Paste this exact code into that box: uniform float4 color = {0, 0, 0, 1}; uniform float width = 2.0; float4 mainImage(VertData v_in) : TARGET { float4 col = image.Sample(textureSampler, v_in.uv); if (col.a > 0.5) return col; float2 thickness = width * uv_pixel_size; for (float x = -1; x <= 1; x++) { for (float y = -1; y <= 1; y++) { if (x == 0 && y == 0) continue; float4 neighbor = image.Sample(textureSampler, v_in.uv + float2(x, y) * thickness); if (neighbor.a > 0.1) return color; } } return col; } Important Checklist for this to work: Extra Pixels: Look at your screenshot.
